/// 缁熶竴鍝嶅簲寮忔崲绠椼€備笟鍔′唬鐮佺敤 context.r / context.sp锛屼笉鍐欐鍍忕礌銆?extension ResponsiveX on BuildContext {
  double get screenWidth => MediaQuery.sizeOf(this).width;
  double get screenHeight => MediaQuery.sizeOf(this).height;
  bool get isPhone => screenWidth < 600;
  bool get isTablet => screenWidth >= 600 && screenWidth < 1000;
  bool get isPortrait => screenHeight >= screenWidth;
  bool get isWide => screenWidth >= 600;

  double get scale {
    final tableDevice = MediaQuery.sizeOf(this).shortestSide >= 600;
    final baseWidth = tableDevice ? 1112.0 : (isPortrait ? 390.0 : 844.0);
    final baseHeight = tableDevice ? 834.0 : (isPortrait ? 844.0 : 390.0);
    final s = (screenWidth / baseWidth).clamp(0.0, screenHeight / baseHeight);
    return s.clamp(0.82, 1.08);
  }

  double r(double base) => base * scale;
  double sp(double base) => base * scale;
  double v(double fraction) => screenHeight * fraction;
  double h(double fraction) => screenWidth * fraction;
}

extension ResponsiveBox on Widget {
  Widget padAll(BuildContext context, double value) =>
      Padding(padding: EdgeInsets.all(context.r(value)), child: this);
}

class _MomentsViewState extends ConsumerState<MomentsView> {

  @override
  Widget build(BuildContext context) {
    final liked = ref.watch(likedMomentsProvider);
    final comments = ref.watch(momentCommentsProvider);
    return PhonePage(
      key: const Key('moments_view'),
      header: ScreenHeader(
        title: '鏈嬪弸鍦?,
        subtitle: '浠栫殑鐢熸椿 路 绗?5 澶?,
        actions: [
          AppIconButton(
            icon: AppIconName.camera,
            onTap: () => _showCamera(context),
          ),
        ],
      ),
      child: FadeMask(
        child: ListView.separated(
          padding: EdgeInsets.fromLTRB(
            AppSpacing.pageH(context),
            context.r(62),
            AppSpacing.pageH(context),
            context.isPortrait ? context.r(112) : AppSpacing.lg(context),
          ),
          itemCount: demoMoments.length,
          separatorBuilder: (context, index) =>
              SizedBox(height: AppSpacing.md(context)),
          itemBuilder: (context, index) {
            final item = demoMoments[index];
            return MomentCard(
              moment: item,
              liked: liked.contains(item.id),
              comments: comments[item.id] ?? item.comments,
              onLike: () => _toggle(item.id),
              onTap: () => _open(item),
            );
          },
        ),
      ),
    );
  }

  void _toggle(String id) {
    ref.read(likedMomentsProvider.notifier).update((liked) {
      final next = {...liked};
      if (!next.remove(id)) next.add(id);
      return next;
    });
  }

  void _open(Moment moment) {
    ref.read(momentOverlayProvider.notifier).state = moment;
    showPopup(
      context,
      Builder(
        builder: (popContext) => MomentDetailSheet(
          moment: moment,
          liked: ref.read(likedMomentsProvider).contains(moment.id),
          comments: ref.read(momentCommentsProvider)[moment.id],
          onClose: () => Navigator.of(popContext).pop(),
          onLike: () {
            ref.read(likedMomentsProvider.notifier).update((liked) {
              final next = {...liked};
              if (!next.remove(moment.id)) next.add(moment.id);
              return next;
            });
          },
          onAddComment: (text) {
            ref.read(momentCommentsProvider.notifier).update((value) {
              final next = {...value};
              next[moment.id] = [
                ...?value[moment.id],
                MomentComment(author: '浣?, text: text, time: '鍒氬垰'),
              ];
              return next;
            });
          },
        ),
      ),
    ).whenComplete(() {
      if (mounted) ref.read(momentOverlayProvider.notifier).state = null;
    });
  }

  void _showCamera(BuildContext context) {
    showPhotoPicker(context);
  }
}

class CompanionShell extends StatelessWidget {
  const CompanionShell({
    super.key,
    required this.location,
    required this.child,
  });

  final String location;
  final Widget child;

  int get _index {
    if (location.startsWith('/moments')) return 1;
    if (location.startsWith('/diary')) return 2;
    if (location.startsWith('/me')) return 3;
    if (location.startsWith('/chat')) return 4;
    return 0;
  }

  void _go(BuildContext context, int index) {
    const paths = ['/today', '/moments', '/diary', '/me', '/chat'];
    context.go(paths[index]);
  }

  @override
  Widget build(BuildContext context) {
    final size = MediaQuery.sizeOf(context);
    final tablet = size.shortestSide >= 600;
    final portrait = size.height >= size.width;
    if (tablet) {
      return Scaffold(
        key: const Key('companion_shell_tablet'),
        backgroundColor: Colors.black,
        body: CinemaTabletView(
          activeIndex: _index,
          chatOpen: location.startsWith('/chat'),
          onSelect: (i) => _go(context, i),
          onChat: () => context.go('/chat'),
          onCloseChat: () => context.go('/today'),
        ),
      );
    }

    final chat = location.startsWith('/chat');
    final rightWidth = size.width * 0.37;
    final bottomInset = MediaQuery.paddingOf(context).bottom;
    return Scaffold(
      key: const Key('companion_shell_phone'),
      backgroundColor: Colors.black,
      resizeToAvoidBottomInset: true,
      body: Stack(
        children: [
          Positioned.fill(
            child: RepaintBoundary(
              child: Image.asset('assets/media/bg.png', fit: BoxFit.cover),
            ),
          ),
          Positioned.fill(
            left: !chat && !portrait ? context.r(58) : 0,
            right: !chat && !portrait ? rightWidth : 0,
            child: child,
          ),
          if (!chat && !portrait)
            Positioned(
              top: 0,
              right: 0,
              bottom: 0,
              width: rightWidth,
              child: const SafeArea(
                top: true,
                bottom: false,
                child: PhoneRightPanel(),
              ),
            ),
          if (!chat && !portrait)
            Positioned(
              left: 0,
              top: 0,
              bottom: 0,
              child: SafeArea(
                child: Center(
                  child: SideDock(
                    index: _index,
                    onSelect: (i) => _go(context, i),
                  ),
                ),
              ),
            ),
          if (!chat && portrait)
            Positioned(
              left: 0,
              right: 0,
              bottom: bottomInset + context.r(12),
              child: Center(
                child: Consumer(
                  builder: (context, ref, child) {
                    final sheetOpen = ref.watch(appSheetOpenProvider);
                    if (sheetOpen) return const SizedBox.shrink();
                    return Dock(
                      index: _index,
                      onSelect: (i) => _go(context, i),
                    );
                  },
                ),
              ),
            ),
        ],
      ),
    );
}
}

  @override

}

class _CinemaChatOverlayState extends State<CinemaChatOverlay> {
  final _controller = TextEditingController();
  final _scroll = ScrollController();
  late final List<ChatMessage> _messages = [...demoChat];

  @override
  void dispose() {
    _controller.dispose();
    _scroll.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Positioned(
      right: context.h(0.025),
      top: context.h(0.09),
      bottom: context.h(0.03),
      width: context.h(0.33),
      child: GlassPanel(
        padding: EdgeInsets.zero,
        child: Column(
          children: [
            _ChatOverlayHeader(onClose: widget.onClose),
            Expanded(
              child: ListView.separated(
                controller: _scroll,
                padding: EdgeInsets.all(AppSpacing.sm(context)),
                itemCount: _messages.length,
                separatorBuilder: (context, index) =>
                    SizedBox(height: AppSpacing.xs(context)),
                itemBuilder: (context, index) =>
                    _MiniBubble(message: _messages[index]),
              ),
            ),
            _OverlayComposer(controller: _controller, onSend: _send),
          ],
        ),
      ),
    );
  }

  void _send() {
    final text = _controller.text.trim();
    if (text.isEmpty) return;
    setState(() {
      _messages.add(
        ChatMessage(
          id: 'local-${DateTime.now().millisecondsSinceEpoch}',
          text: text,
          fromCompanion: false,
          read: true,
        ),
      );
      _controller.clear();
    });
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (_scroll.hasClients) _scroll.jumpTo(_scroll.position.maxScrollExtent);
    });
  }
}

class AppIcon extends StatelessWidget {
  const AppIcon(
    this.name, {
    super.key,
    required this.size,
    this.color = Colors.white,
    this.strokeWidth,
  });

  final AppIconName name;
  final double size;
  final Color color;
  final double? strokeWidth;

  @override
  Widget build(BuildContext context) {
    return CustomPaint(
      size: Size.square(size),
      painter: _IconPainter(
        name: name,
        color: color,
        strokeWidth: strokeWidth ?? size * 0.075,
      ),
      isComplex: true,
    );
  }
}

class _IconPainter extends CustomPainter {
  _IconPainter({
    required this.name,
    required this.color,
    required this.strokeWidth,
  });

  final AppIconName name;
  final Color color;
  final double strokeWidth;

  @override
  void paint(Canvas canvas, Size size) {
    final scale = size.width / 24;
    canvas.scale(scale);
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth / scale
      ..strokeCap = StrokeCap.round
      ..strokeJoin = StrokeJoin.round;
    final fill = Paint()
      ..color = color
      ..style = PaintingStyle.fill;
    switch (name) {
      case AppIconName.sun:
        canvas.drawCircle(const Offset(12, 12), 4, paint);
        for (var i = 0; i < 8; i++) {
          final angle = i * math.pi / 4;
          canvas.drawLine(
            Offset(12 + math.cos(angle) * 7.2, 12 + math.sin(angle) * 7.2),
            Offset(12 + math.cos(angle) * 9.6, 12 + math.sin(angle) * 9.6),
            paint,
          );
        }
      case AppIconName.moments:
        canvas.drawRRect(
          RRect.fromRectAndRadius(
            const Rect.fromLTWH(3, 3, 18, 18),
            const Radius.circular(4),
          ),
          paint,
        );
        canvas.drawPath(
          Path()
            ..moveTo(3, 16)
            ..lineTo(8, 10.5)
            ..lineTo(12.5, 15)
            ..lineTo(16, 12)
            ..lineTo(21, 17),
          paint,
        );
        canvas.drawCircle(const Offset(16, 8), 1.4, fill);
      case AppIconName.diary:
        canvas.drawRRect(
          RRect.fromRectAndRadius(
            const Rect.fromLTWH(4, 3, 16, 18),
            const Radius.circular(3),
          ),
          paint,
        );
        canvas.drawLine(const Offset(4, 9), const Offset(20, 9), paint);
        canvas.drawLine(const Offset(8, 2.5), const Offset(8, 5.5), paint);
        canvas.drawLine(const Offset(16, 2.5), const Offset(16, 5.5), paint);
        canvas.drawLine(const Offset(8, 13), const Offset(16, 13), paint);
      case AppIconName.person:
        canvas.drawCircle(const Offset(12, 8), 3.5, paint);
        canvas.drawPath(
          Path()
            ..moveTo(4.5, 20)
            ..cubicTo(5.2, 16.3, 8.2, 14.3, 12, 14.3)
            ..cubicTo(15.8, 14.3, 18.8, 16.3, 19.5, 20),
          paint,
        );
      case AppIconName.chat:
        canvas.drawRRect(
          RRect.fromRectAndRadius(
            const Rect.fromLTWH(3, 4, 18, 13),
            const Radius.circular(5),
          ),
          paint,
        );
        canvas.drawPath(
          Path()
            ..moveTo(8, 16)
            ..lineTo(6.5, 20.5)
            ..lineTo(11.5, 17),
          paint,
        );
      case AppIconName.search:
        canvas.drawCircle(const Offset(10.5, 10.5), 6, paint);
        canvas.drawLine(const Offset(15.5, 15.5), const Offset(21, 21), paint);
      case AppIconName.camera:
        canvas.drawRRect(
          RRect.fromRectAndRadius(
            const Rect.fromLTWH(3, 6, 18, 13),
            const Radius.circular(3),
          ),
          paint,
        );
        canvas.drawCircle(const Offset(12, 12.5), 3.5, paint);
        canvas.drawLine(const Offset(8, 6), const Offset(9.2, 4), paint);
        canvas.drawLine(const Offset(9.2, 4), const Offset(14.8, 4), paint);
        canvas.drawLine(const Offset(14.8, 4), const Offset(16, 6), paint);
      case AppIconName.more:
        for (final x in <double>[3.5, 12, 20.5]) {
          canvas.drawCircle(Offset(x, 12), 1.5, fill);
        }
      case AppIconName.grid:
        for (final rect in [
          const Rect.fromLTWH(3.5, 3.5, 7, 7),
          const Rect.fromLTWH(13.5, 3.5, 7, 7),
          const Rect.fromLTWH(3.5, 13.5, 7, 7),
          const Rect.fromLTWH(13.5, 13.5, 7, 7),
        ]) {
          canvas.drawRRect(
            RRect.fromRectAndRadius(rect, const Radius.circular(2)),
            paint,
          );
        }
      case AppIconName.settings:
        for (var i = 0; i < 3; i++) {
          final y = 6.0 + i * 6;
          canvas.drawLine(Offset(3, y), Offset(21, y), paint);
          canvas.drawCircle(Offset(i == 1 ? 14 : 8, y), 2, fill);
        }
      case AppIconName.book:
        canvas.drawPath(
          Path()
            ..moveTo(4, 5)
            ..cubicTo(7, 3.5, 10, 4.5, 12, 6)
            ..cubicTo(14, 4.5, 17, 3.5, 20, 5)
            ..lineTo(20, 19)
            ..cubicTo(17, 17.5, 14, 18.5, 12, 20)
            ..cubicTo(10, 18.5, 7, 17.5, 4, 19)
            ..close(),
          paint,
        );
        canvas.drawLine(const Offset(12, 6), const Offset(12, 20), paint);
      case AppIconName.calendar:
        canvas.drawRRect(
          RRect.fromRectAndRadius(
            const Rect.fromLTWH(4, 5, 16, 16),
            const Radius.circular(3),
          ),
          paint,
        );
        canvas.drawLine(const Offset(4, 10), const Offset(20, 10), paint);
        canvas.drawLine(const Offset(8, 2.5), const Offset(8, 6), paint);
        canvas.drawLine(const Offset(16, 2.5), const Offset(16, 6), paint);
      case AppIconName.heart:
        canvas.drawPath(
          Path()
            ..moveTo(12, 20)
            ..cubicTo(10.5, 18.5, 4, 14, 4, 8.7)
            ..cubicTo(4, 5.6, 7.2, 3.7, 9.7, 5.8)
            ..lineTo(12, 7.8)
            ..lineTo(14.3, 5.8)
            ..cubicTo(16.8, 3.7, 20, 5.6, 20, 8.7)
            ..cubicTo(20, 14, 13.5, 18.5, 12, 20)
            ..close(),
          paint,
        );
      case AppIconName.chevron:
        canvas.drawPath(
          Path()
            ..moveTo(9, 5)
            ..lineTo(16, 12)
            ..lineTo(9, 19),
          paint,
        );
      case AppIconName.back:
        canvas.drawPath(
          Path()
            ..moveTo(15, 5)
            ..lineTo(8, 12)
            ..lineTo(15, 19),
          paint,
        );
      case AppIconName.plus:
        canvas.drawLine(const Offset(12, 4), const Offset(12, 20), paint);
        canvas.drawLine(const Offset(4, 12), const Offset(20, 12), paint);
      case AppIconName.send:
        canvas.drawPath(
          Path()
            ..moveTo(4, 12)
            ..lineTo(20, 4)
            ..lineTo(14, 20)
            ..lineTo(11.5, 13.5)
            ..close(),
          paint,
        );
      case AppIconName.close:
        canvas.drawLine(const Offset(6, 6), const Offset(18, 18), paint);
        canvas.drawLine(const Offset(18, 6), const Offset(6, 18), paint);
      case AppIconName.thumb:
        canvas.drawPath(
          Path()
            ..moveTo(7, 10)
            ..lineTo(7, 20)
            ..lineTo(3, 20)
            ..lineTo(3, 10)
            ..close(),
          paint,
        );
        canvas.drawPath(
          Path()
            ..moveTo(7, 10)
            ..lineTo(11, 3)
            ..cubicTo(13, 3, 14, 5, 13, 8)
            ..lineTo(20, 8)
            ..cubicTo(21.5, 8, 21.5, 11, 19.5, 11)
            ..lineTo(17, 11)
            ..lineTo(20, 18)
            ..cubicTo(20.5, 20, 18, 21, 16, 20)
            ..lineTo(10, 18),
          paint,
        );
    }
  }

  @override
  bool shouldRepaint(covariant _IconPainter oldDelegate) =>
      oldDelegate.name != name ||
      oldDelegate.color != color ||
      oldDelegate.strokeWidth != strokeWidth;
}

Future<void> showAppSheet(
  BuildContext context, {
  required String title,
  required Widget child,
  bool full = false,
  double heightFactor = 0.55,
}) {
  final host = AppSheetHost.hostKey.currentState;
  final body = _AppSheetBody(
    title: title,
    full: full,
    heightFactor: heightFactor,
    child: child,
  );
  if (host != null) return host.show(body);
  return showModalBottomSheet<void>(
    context: context,
    backgroundColor: Colors.transparent,
    isScrollControlled: true,
    useSafeArea: false,
    builder: (context) => body,
  );
}

final appSheetOpenProvider = StateProvider<bool>((ref) => false);

Future<void> showPopup(
  BuildContext context,
  Widget child, {
  bool dismissible = true,
}) async {
  final host = AppSheetHost.hostKey.currentState;
  if (host == null) return;
  await host.showRaw(child, dismissible: dismissible);
}

  const AppSheetHost({super.key});

  static final GlobalKey<AppSheetHostState> hostKey =
      GlobalKey<AppSheetHostState>();

  @override
  State<AppSheetHost> createState() => AppSheetHostState();
}

class AppSheetHostState extends State<AppSheetHost> {
  final _navigator = GlobalKey<NavigatorState>();
  bool _open = false;

  Future<void> showRaw(
    Widget child, {
    bool dismissible = false,
  }) async {
    final nav = _navigator.currentState;
    if (nav == null) return;
    ProviderScope.containerOf(context)
        .read(appSheetOpenProvider.notifier)
        .state = true;
    setState(() => _open = true);
    final route = PageRouteBuilder<void>(
      opaque: false,
      barrierColor: const Color(0x8C000000),
      barrierDismissible: dismissible,
      transitionDuration: const Duration(milliseconds: 180),
      reverseTransitionDuration: const Duration(milliseconds: 160),
      pageBuilder: (context, animation, secondaryAnimation) => child,
    );
    await nav.push(route);
    if (mounted) {
      ProviderScope.containerOf(context)
          .read(appSheetOpenProvider.notifier)
          .state = false;
      setState(() => _open = false);
    }
  }

  Future<void> show(Widget sheet) async {
    final nav = _navigator.currentState;
    if (nav == null) return;
    ProviderScope.containerOf(context)
        .read(appSheetOpenProvider.notifier)
        .state = true;
    setState(() => _open = true);
    final route = PageRouteBuilder<void>(
      opaque: false,
      barrierColor: const Color(0x85000000),
      transitionDuration: const Duration(milliseconds: 180),
      reverseTransitionDuration: const Duration(milliseconds: 160),

class ScreenHeader extends StatelessWidget {
  const ScreenHeader({
    super.key,
    required this.title,
    this.subtitle,
    this.leading,
    this.actions = const [],
    this.compact = false,
  });

  final String title;
  final String? subtitle;
  final Widget? leading;
  final List<Widget> actions;
  final bool compact;

  @override
  Widget build(BuildContext context) {
    return GlassPanel(
      key: const Key('screen_header'),
      padding: EdgeInsets.symmetric(
        horizontal: context.isPortrait ? context.r(10) : AppSpacing.md(context),
        vertical: 0,
      ),
      borderRadius: BorderRadius.circular(AppRadius.lg(context)),
      child: SizedBox(
        height: context.r(compact ? 38 : 46),
        child: Row(
          children: [
            if (leading != null) ...[
              leading!,
              SizedBox(width: AppSpacing.xs(context)),
            ],
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Text(
                    title,
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: AppTextStyles.bodyStrong(
                      context,
                    ).copyWith(fontSize: context.sp(14)),
                  ),
                  if (subtitle != null)
                    Text(
                      subtitle!,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: AppTextStyles.caption(
                        context,
                      ).copyWith(fontSize: context.sp(8.5)),
                    ),
                ],
              ),
            ),
            for (final action in actions) ...[
              SizedBox(width: AppSpacing.xxs(context)),
              action,
            ],
          ],
        ),
      ),
    );
  }
}

  final String label;

const sideDockItems = [
  SideDockItem(label: '浠婂ぉ', icon: AppIconName.sun),
  SideDockItem(label: '鏈嬪弸鍦?, icon: AppIconName.moments),
  SideDockItem(label: '鏃ヨ', icon: AppIconName.diary),
  SideDockItem(label: '浠?, icon: AppIconName.person),
  SideDockItem(label: '瀵硅瘽', icon: AppIconName.chat),

class SideDock extends StatelessWidget {
  const SideDock({super.key, required this.index, required this.onSelect});

  final int index;
  final ValueChanged<int> onSelect;

  @override
  Widget build(BuildContext context) {
    final radius = BorderRadius.circular(context.r(27));
    return ClipRRect(
      borderRadius: radius,
      child: DecoratedBox(
        key: const Key('phone_side_dock'),
        decoration: BoxDecoration(
          color: const Color(0xF21B1411),
          borderRadius: radius,
          border: Border.all(color: AppColors.border),
        ),
        child: Padding(
          padding: EdgeInsets.symmetric(
            horizontal: context.r(4),
            vertical: context.r(6),
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              for (var i = 0; i < sideDockItems.length; i++) ...[
                if (i > 0) SizedBox(height: context.r(2)),
                _DockItem(
                  item: sideDockItems[i],
                  selected: i == index,
                  onTap: () => onSelect(i),
                ),
              ],
            ],
          ),
        ),
      ),
    );
  }
}

class _DockItem extends StatelessWidget {
  const _DockItem({
    required this.item,
    required this.selected,
    required this.onTap,
  });

  final SideDockItem item;
  final bool selected;
  final VoidCallback onTap;

  @override
  Widget build(BuildContext context) {
    final radius = BorderRadius.circular(context.r(12));
    return GestureDetector(
      behavior: HitTestBehavior.opaque,
      onTap: onTap,
      child: DecoratedBox(
        decoration: BoxDecoration(
          borderRadius: radius,
          gradient: selected
              ? const LinearGradient(
                  begin: Alignment.topCenter,
                  end: Alignment.bottomCenter,
                  colors: [Color(0x33FFFFFF), Color(0x0FFFFFFF)],
                )
              : null,
          border: selected ? Border.all(color: const Color(0x29FFFFFF)) : null,
          boxShadow: selected
              ? const [
                  BoxShadow(
                    color: Color(0x3D000000),
                    blurRadius: 14,
                    offset: Offset(0, 4),
                  ),
                ]
              : null,
        ),
        child: SizedBox(
          width: context.r(46),
          height: context.r(46),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              AppIcon(
                item.icon,
                size: context.r(18),
                color: selected ? AppColors.ink : AppColors.inkFaint,
              ),
              SizedBox(height: context.r(2)),
              Text(
                item.label,
                maxLines: 1,
                overflow: TextOverflow.ellipsis,
                style: AppTextStyles.micro(context).copyWith(
                  letterSpacing: 0.02,
                  color: selected ? AppColors.ink : AppColors.inkFaint,
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
